Crear una BASE INICIAL única para David. -- Se calcula solo una vez para que el total actual quede en 3000 XP. -- --------------------------------------------------------------------- with actual as ( select coalesce(sum(p.puntos), 0)::integer as total_actual from public.puntos_eventos p where p.integrante_id = 1 ), sesion as ( select s.id from public.estudio_sesiones s where s.legacy_id = 7 or s.codigo = 'EST-007' order by case when s.codigo = 'EST-007' then 0 else 1 end limit 1 ) insert into public.puntos_eventos ( integrante_id, sesion_id, tipo, puntos, motivo, referencia, metadata ) select 1, s.id, 'ajuste', 3000 - a.total_actual, 'Base inicial de XP por liderazgo y participación acumulada', 'BASE-XP-DAVID-3000', jsonb_build_object( 'origen', 'base_inicial_administrativa', 'base_objetivo_al_momento_de_creacion', 3000, 'total_previo', a.total_actual, 'nota', 'La base no es un techo. Crear una BASE INICIAL única para Oneiber. -- --------------------------------------------------------------------- with actual as ( select coalesce(sum(p.puntos), 0)::integer as total_actual from public.puntos_eventos p where p.integrante_id = 575 ), sesion as ( select s.id from public.estudio_sesiones s where s.legacy_id = 7 or s.codigo = 'EST-007' order by case when s.codigo = 'EST-007' then 0 else 1 end limit 1 ) insert into public.puntos_eventos ( integrante_id, sesion_id, tipo, puntos, motivo, referencia, metadata ) select 575, s.id, 'ajuste', 2000 - a.total_actual, 'Base inicial de XP por liderazgo y participación acumulada', 'BASE-XP-ONEIBER-2000', jsonb_build_object( 'origen', 'base_inicial_administrativa', 'base_objetivo_al_momento_de_creacion', 2000, 'total_previo', a.total_actual, 'nota', 'La base no es un techo. Los XP posteriores se suman normalmente.' ) from actual a cross join sesion s where (2000 - a.total_actual) <> 0 and not exists ( select 1 from public.puntos_eventos p where p.referencia = 'BASE-XP-ONEIBER-2000' ); commit; -- --------------------------------------------------------------------- -- 4. Verificación -- --------------------------------------------------------------------- with totales as ( select p.integrante_id, sum(p.puntos)::bigint as puntos from public.puntos_eventos p where p.integrante_id in (1, 575) group by p.integrante_id ) select t.integrante_id, trim(concat_ws(' ', i.nombres, i.apellidos)) as nombre, t.puntos as xp_total_actual, coalesce( ( select n.icono from public.ranking_niveles n where n.xp_min <= t.puntos order by n.xp_min desc limit 1 ), '🥉' ) as icono, coalesce( ( select n.nombre from public.ranking_niveles n where n.xp_min <= t.puntos order by n.xp_min desc limit 1 ), 'Participante' ) as nivel from totales t join public.integrantes i on i.id = t.integrante_id order by t.puntos desc;
